===== examples/nested_if_else.src =====
a = 10;
b = 20;
c = 5;

if (a < b) {
    if (b > c) {
        result = 1;
    } else {
        result = 2;
    }
} else {
    if (a == c) {
        result = 3;
    } else {
        result = 4;
    }
}
done = result;
--- output ---
Three Address Code:
1: a = 10
2: b = 20
3: c = 5
4: if a < b goto L1
5: goto L2
L1: if b > c goto L3
7: goto L4
L3: result = 1
9: goto L5
L4: result = 2
11: goto L5
L2: if a == c goto L6
13: goto L7
L6: result = 3
15: goto L5
L7: result = 4
L5: done = result
18: end
exit code: 0
